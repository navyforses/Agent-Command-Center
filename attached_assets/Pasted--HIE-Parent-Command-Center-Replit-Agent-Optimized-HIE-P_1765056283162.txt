### 1️⃣ Document Analysis Agent

```typescript
// lib/ai/agents/documentAgent.ts

import { ChatOpenAI } from "@langchain/openai";
import { PromptTemplate } from "@langchain/core/prompts";
import { StructuredOutputParser } from "langchain/output_parsers";
import { z } from "zod";

const documentAnalysisSchema = z.object({
  documentType: z.string(),
  keyFindings: z.array(z.object({
    finding: z.string(),
    severity: z.enum(["normal", "mild", "moderate", "severe"]),
    location: z.string().optional(),
  })),
  diagnoses: z.array(z.string()),
  measurements: z.array(z.object({
    name: z.string(),
    value: z.string(),
    unit: z.string().optional(),
    normalRange: z.string().optional(),
    isAbnormal: z.boolean(),
  })),
  recommendations: z.array(z.string()),
  followUpNeeded: z.boolean(),
  summaryForParent: z.string(),
  summaryForParentKa: z.string(), // ქართულად
});

export class DocumentAnalysisAgent {
  private model: ChatOpenAI;
  private parser: StructuredOutputParser<typeof documentAnalysisSchema>;

  constructor() {
    this.model = new ChatOpenAI({
      modelName: "gpt-4-turbo-preview",
      temperature: 0.1,
    });
    this.parser = StructuredOutputParser.fromZodSchema(documentAnalysisSchema);
  }

  async analyze(ocrText: string, documentType: string): Promise<z.infer<typeof documentAnalysisSchema>> {
    const prompt = PromptTemplate.fromTemplate(`
შენ ხარ პედიატრიული ნევროლოგიის ექსპერტი, რომელიც აანალიზებს სამედიცინო დოკუმენტებს HIE-ით დაავადებული ბავშვებისთვის.

დოკუმენტის ტიპი: {documentType}
დოკუმენტის ტექსტი:
{ocrText}

გააანალიზე ეს დოკუმენტი და ამოიღე შემდეგი ინფორმაცია:
1. ძირითადი აღმოჩენები (findings) სიმძიმის მითითებით
2. დიაგნოზები
3. გაზომვები და მათი ნორმალური დიაპაზონები
4. რეკომენდაციები
5. საჭიროა თუ არა follow-up

ასევე დაწერე მარტივი შეჯამება მშობლისთვის ინგლისურად და ქართულად.

{format_instructions}
    `);

    const formattedPrompt = await prompt.format({
      documentType,
      ocrText,
      format_instructions: this.parser.getFormatInstructions(),
    });

    const response = await this.model.invoke(formattedPrompt);
    return this.parser.parse(response.content as string);
  }
}
```

```typescript
// lib/ai/agents/trialAgent.ts

import { ChatOpenAI } from "@langchain/openai";

interface TrialSearchParams {
  condition: string;
  ageMonths: number;
  location?: string;
  interventionType?: string;
}

interface EligibilityResult {
  trialId: string;
  title: string;
  matchScore: number;
  matchedCriteria: string[];
  unmatchedCriteria: string[];
  uncertainCriteria: string[];
  recommendation: string;
  contactInfo: any;
}

export class ClinicalTrialAgent {
  private model: ChatOpenAI;

  constructor() {
    this.model = new ChatOpenAI({
      modelName: "gpt-4-turbo-preview",
      temperature: 0.1,
    });
  }

  async searchTrials(params: TrialSearchParams): Promise<any[]> {
    // ClinicalTrials.gov API call
    const baseUrl = "https://clinicaltrials.gov/api/v2/studies";
    const query = new URLSearchParams({
      "query.cond": params.condition,
      "filter.overallStatus": "RECRUITING,NOT_YET_RECRUITING",
      "pageSize": "50",
    });

    const response = await fetch(`${baseUrl}?${query}`);
    const data = await response.json();
    
    return data.studies || [];
  }

  async checkEligibility(trial: any, childProfile: any): Promise<EligibilityResult> {
    const prompt = `
გააანალიზე კლინიკური კვლევის ელიგიბელობის კრიტერიუმები და შეადარე ბავშვის პროფილს.

## კვლევა:
სათაური: ${trial.protocolSection?.identificationModule?.briefTitle}
ID: ${trial.protocolSection?.identificationModule?.nctId}

ელიგიბელობის კრიტერიუმები:
${trial.protocolSection?.eligibilityModule?.eligibilityCriteria}

ასაკის დიაპაზონი: ${trial.protocolSection?.eligibilityModule?.minimumAge} - ${trial.protocolSection?.eligibilityModule?.maximumAge}

## ბავშვის პროფილი:
${JSON.stringify(childProfile, null, 2)}

შეაფასე:
1. რამდენად შეესაბამება ბავშვი კრიტერიუმებს (0-100%)
```

```typescript
function calculateAge(dateOfBirth: Date): string {
  const now = new Date();
  const birth = new Date(dateOfBirth);
  const months = (now.getFullYear() - birth.getFullYear()) * 12 + (now.getMonth() - birth.getMonth());
  
  if (months < 24) {
    return `${months} თვე`;
  }
  return `${Math.floor(months / 12)} წელი`;
}
```

```typescript
// app/api/documents/upload/route.ts

import { NextRequest, NextResponse } from "next/server";
import { v2 as cloudinary } from "cloudinary";
import Tesseract from "tesseract.js";
import { DocumentAnalysisAgent } from "@/lib/ai/agents/documentAgent";
import { prisma } from "@/lib/prisma";

cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET,
});

export async function POST(req: NextRequest) {
  try {
    const formData = await req.formData();
    const file = formData.get("file") as File;
    const childId = formData.get("childId") as string;
    const documentType = formData.get("documentType") as string;

    // 1. Upload to Cloudinary
    const buffer = Buffer.from(await file.arrayBuffer());
    const uploadResult = await new Promise((resolve, reject) => {
      cloudinary.uploader.upload_stream(
        { folder: "hie-documents", resource_type: "auto" },
        (error, result) => {
          if (error) reject(error);
          else resolve(result);
        }
      ).end(buffer);
    });

    // 2. OCR Processing
    let ocrText = "";
    let ocrConfidence = 0;

    if (file.type.includes("image") || file.type === "application/pdf") {
      const ocrResult = await Tesseract.recognize(buffer, "eng+fra+kat", {
        logger: (m) => console.log(m),
      });
      ocrText = ocrResult.data.text;
      ocrConfidence = ocrResult.data.confidence;
    }

    // 3. AI Analysis
    const documentAgent = new DocumentAnalysisAgent();
    const analysis = await documentAgent.analyze(ocrText, documentType);

    // 4. Save to Database
    const document = await prisma.document.create({
      data: {
        childId,
        fileName: file.name,
        fileType: file.type,
        fileUrl: (uploadResult as any).secure_url,
        fileSize: file.size,
        documentType,
        ocrText,
        ocrConfidence,
        aiSummary: analysis.summaryForParent,
        aiSummaryKa: analysis.summaryForParentKa,
        extractedData: analysis,
        isProcessed: true,
      },
    });

    return NextResponse.json({ success: true, document });
  } catch (error) {
    console.error("Upload error:", error);
    return NextResponse.json({ error: "Upload failed" }, { status: 500 });
  }
}
```

```typescript
// app/api/ai/therapy-recommendations/route.ts

import { NextRequest, NextResponse } from "next/server";
import { TherapyRecommendationAgent } from "@/lib/ai/agents/therapyAgent";
import { prisma } from "@/lib/prisma";

export async function POST(req: NextRequest) {
  try {
    const { childId } = await req.json();

    // Get child profile with all related data
    const child = await prisma.child.findUnique({
      where: { id: childId },
      include: {
        documents: {
          where: { isProcessed: true },
          orderBy: { createdAt: "desc" },
        },
        therapies: {
          where: { isActive: true },
        },
      },
    });

    if (!child) {
      return NextResponse.json({ error: "Child not found" }, { status: 404 });
    }

    // Generate recommendations
    const therapyAgent = new TherapyRecommendationAgent();
    const recommendations = await therapyAgent.generateRecommendations(
      child,
      child.documents,
      child.therapies
    );

    // Save to database
    const savedRecommendation = await prisma.aIRecommendation.create({
      data: {
        childId,
        recommendations,
        priorityList: recommendations.priorityTherapies,
        actionPlan: recommendations.actionPlan,
        basedOnDocs: child.documents.map((d) => d.id),
        modelUsed: "gpt-4-turbo + claude-3-opus",
        reviewDate: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000), // 30 days
      },
    });

    return NextResponse.json({
      success: true,
      recommendation: savedRecommendation,
    });
  } catch (error) {
    console.error("Recommendation error:", error);
    return NextResponse.json(
      { error: "Failed to generate recommendations" },
      { status: 500 }
    );
  }
}
```
