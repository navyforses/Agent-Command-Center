# 🌐 NEXUS OMEGA - Replit Development Prompt
## Multi-AI Neuroregeneration Research Platform

---

## 📋 PROJECT OVERVIEW

Create a new page/module called **"NEXUS OMEGA"** for our existing HIE Parent Command Center web application. This is an advanced AI-powered research platform that coordinates multiple AI systems (Claude, ChatGPT, Grok, Gemini, Perplexity) to conduct comprehensive neuroregeneration research with cross-disciplinary scientific analysis.

**Purpose:** Help parents of children with severe brain injuries (like HIE - Hypoxic-Ischemic Encephalopathy) find cutting-edge treatments by leveraging collective AI intelligence and all scientific disciplines.

**Patient Context:** Alexandra - infant with severe neonatal HIE with extensive brain damage.

## 💾 DATABASE SCHEMA

```sql
-- AI Agents
CREATE TABLE ai_agents (
  id VARCHAR(50) PRIMARY KEY,
  name VARCHAR(100),
  color VARCHAR(7),
  description TEXT,
  strengths TEXT[],
  created_at TIMESTAMP DEFAULT NOW()
);

-- Research Queries
CREATE TABLE research_queries (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  query_text TEXT NOT NULL,
  disciplines TEXT[],
  research_focus TEXT[],
  status VARCHAR(50),
  created_at TIMESTAMP DEFAULT NOW(),
  completed_at TIMESTAMP
);

-- Findings
CREATE TABLE findings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  query_id UUID REFERENCES research_queries(id),
  title TEXT NOT NULL,
  summary TEXT,
  consensus_level VARCHAR(10),
  confidence_score INTEGER,
  relevance_score INTEGER,
  sources JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- AI Analyses
CREATE TABLE ai_analyses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  finding_id UUID REFERENCES findings(id),
  ai_agent_id VARCHAR(50) REFERENCES ai_agents(id),
  perspective TEXT,
  confidence INTEGER,
  key_points TEXT[],
  concerns TEXT[],
  unique_insights TEXT[],
  raw_response JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Disciplinary Analyses
CREATE TABLE disciplinary_analyses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  finding_id UUID REFERENCES findings(id),
  discipline VARCHAR(100),
  analysis TEXT,
  cross_connections JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Knowledge Nodes
CREATE TABLE knowledge_nodes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  label TEXT NOT NULL,
  node_type VARCHAR(50),
  relevance_score INTEGER,
  evidence_level INTEGER,
  metadata JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Knowledge Edges
CREATE TABLE knowledge_edges (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  source_id UUID REFERENCES knowledge_nodes(id),
  target_id UUID REFERENCES knowledge_nodes(id),
  relationship VARCHAR(100),
  strength FLOAT,
  discovered_by TEXT[],
  created_at TIMESTAMP DEFAULT NOW()
);

-- Hypotheses
CREATE TABLE hypotheses (
  id VARCHAR(20) PRIMARY KEY,  -- H-001, H-002, etc.
  statement TEXT NOT NULL,
  status VARCHAR(50),
  confidence_score INTEGER,
  proposed_by VARCHAR(50),
  supported_by TEXT[],
  supporting_evidence JSONB,
  contradicting_evidence JSONB,
  cross_disciplinary_basis JSONB,
  testability TEXT,
  action_items JSONB,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Debates
CREATE TABLE debates (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  question TEXT NOT NULL,
  position_a JSONB,
  position_b JSONB,
  resolution_needed TEXT,
  priority VARCHAR(20),
  status VARCHAR(50),
  created_at TIMESTAMP DEFAULT NOW()
);

-- Action Items
CREATE TABLE action_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  description TEXT,
  priority VARCHAR(20),
  source_type VARCHAR(50),  -- 'finding', 'hypothesis', 'debate'
  source_id UUID,
  status VARCHAR(50),
  due_date DATE,
  created_at TIMESTAMP DEFAULT NOW(),
  completed_at TIMESTAMP
);
```

### Chat Service Implementation

```typescript
class ReportChatService {
  
  /**
   * Process user question about a report
   */
  async askQuestion(
    reportId: string, 
    userMessage: string,
    options?: { streamResponse?: boolean }
  ): Promise<ChatMessage> {
    
    // 1. Load report context
    const report = await this.loadReportWithContext(reportId);
    
    // 2. Build context-aware prompt
    const systemPrompt = this.buildChatSystemPrompt(report);
    
    // 3. Find relevant sections of report using semantic search
    const relevantSections = await this.findRelevantSections(
      report, 
      userMessage
    );
    
    // 4. Query multiple AIs with report context
    const aiResponses = await this.queryAIsWithContext(
      userMessage,
      systemPrompt,
      relevantSections
    );
    
    // 5. Synthesize responses
    const synthesizedResponse = this.synthesizeResponses(aiResponses);
    
    // 6. Extract references
    const references = this.extractReferences(
      synthesizedResponse,
      report
    );
    
    // 7. Save and return
    const chatMessage: ChatMessage = {
      id: generateId(),
      reportId,
      role: 'assistant',
      content: synthesizedResponse.content,
      referencedFindings: references.findings,
      referencedHypotheses: references.hypotheses,
      referencedSources: references.sources,
      respondingAIs: aiResponses.map(r => r.aiId),
      confidence: synthesizedResponse.confidence,
      timestamp: new Date()
    };
    
    await this.saveChatMessage(chatMessage);
    return chatMessage;
  }
  
  /**
   * Build system prompt with report context
   */
  private buildChatSystemPrompt(report: DailyReport): string {
    return `
შენ ხარ NEXUS OMEGA - ნეირორეგენერაციის კვლევითი AI სისტემა.

მომხმარებელი (შაკო) კითხვებს სვამს დღის ანგარიშის შესახებ.

3. აღნიშნე რომელი AI-ები ადასტურებენ ინფორმაციას

5. ალექსანდრას კონტექსტი ყოველთვის გაითვალისწინე

  /**
   * Find relevant sections using semantic search
   */
  private async findRelevantSections(
    report: DailyReport,
    query: string
  ): Promise<RelevantSection[]> {
    // Use embeddings to find most relevant parts
    const queryEmbedding = await this.getEmbedding(query);
    
    const sections = [
      ...report.findings.map(f => ({ type: 'finding', content: f, id: f.id })),
      ...report.hypotheses.map(h => ({ type: 'hypothesis', content: h, id: h.id })),
      ...report.debates.map(d => ({ type: 'debate', content: d, id: d.id })),
    ];
    
    // Rank by similarity
    const ranked = sections.map(section => ({
      ...section,
      similarity: cosineSimilarity(queryEmbedding, section.content.embedding)
    })).sort((a, b) => b.similarity - a.similarity);
    
    return ranked.slice(0, 5);  // Top 5 most relevant
  }
  
  /**
   * Query multiple AIs with context
   */
  private async queryAIsWithContext(
    question: string,
    systemPrompt: string,
    relevantSections: RelevantSection[]
  ): Promise<AIResponse[]> {
    
    const contextualPrompt = `
${systemPrompt}

📌 ყველაზე რელევანტური სექციები ამ კითხვისთვის:
${relevantSections.map(s => `[${s.type}] ${JSON.stringify(s.content)}`).join('\n')}

❓ მომხმარებლის კითხვა: ${question}
`;
    
    // Query Claude and Gemini for report Q&A (best for grounded responses)
    const responses = await Promise.allSettled([
      this.claude.chat(contextualPrompt),
      this.gemini.chat(contextualPrompt),
    ]);
    
    return responses
      .filter(r => r.status === 'fulfilled')
      .map(r => r.value);
  }
  
  /**
   * Generate contextual suggestions
   */
  async generateSuggestions(reportId: string): Promise<QuickQuestion[]> {
    const report = await this.loadReportWithContext(reportId);
    
    const suggestions: QuickQuestion[] = [];
    
    // Add finding-specific questions
    report.findings.slice(0, 3).forEach((finding, i) => {
      suggestions.push({
        id: `finding-${finding.id}`,
        label: `ახსენი: ${finding.title.substring(0, 30)}...`,
        template: `დეტალურად ამიხსენი Finding #${i+1}: "${finding.title}" - რა არის მთავარი აღმოჩენა?`,
        icon: '📄'
      });
    });
    
    // Add hypothesis questions
    if (report.hypotheses.length > 0) {
      suggestions.push({
        id: 'hypotheses',
        label: 'ახალი ჰიპოთეზები',
        template: `რა ახალი ჰიპოთეზები დაიბადა ამ ანგარიშიდან და რამდენად რეალისტურია ისინი?`,
        icon: '💡'
      });
    }
    
    // Add action item question
    if (report.actionItems.length > 0) {
      suggestions.push({
        id: 'actions',
        label: 'პრაქტიკული ნაბიჯები',
        template: 'რა კონკრეტული ნაბიჯები უნდა გადავდგა ამ ანგარიშის საფუძველზე?',
        icon: '✅'
      });
    }
    
    // Always add these
    suggestions.push(
      {
        id: 'alexandra',
        label: 'ალექსანდრასთვის',
        template: 'ამ აღმოჩენებიდან რა არის ყველაზე იმედისმომცემი ალექსანდრასთვის და რატომ?',
        icon: '🎯'
      },
      {
        id: 'timeline',
        label: 'დროის პროგნოზი',
        template: 'რა არის რეალისტური timeline ამ თერაპიების კლინიკური ხელმისაწვდომობისთვის?',
        icon: '📅'
      }
    );
    
    return suggestions;
  }
}
```

```tsx
import React, { useState, useRef, useEffect } from 'react';
import { Send, Loader2, MessageCircle, Lightbulb, RefreshCw } from 'lucide-react';

interface ReportChatProps {

export const ReportChat: React.FC<ReportChatProps> = ({ reportId, report }) => {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [input, setInput] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [suggestions, setSuggestions] = useState<QuickQuestion[]>([]);
  const messagesEndRef = useRef<HTMLDivElement>(null);
  
  useEffect(() => {
    loadChatHistory();
    loadSuggestions();
  }, [reportId]);
  
  useEffect(() => {
    scrollToBottom();
  }, [messages]);
  
  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };
  
  const loadChatHistory = async () => {
    const response = await fetch(`/api/nexus/reports/${reportId}/chat`);
    const data = await response.json();
    setMessages(data);
  };
  
  const loadSuggestions = async () => {
    const response = await fetch(`/api/nexus/reports/${reportId}/chat/suggestions`);
    const data = await response.json();
    setSuggestions(data);
  };
  
  const sendMessage = async (messageText?: string) => {
    const text = messageText || input;
    if (!text.trim()) return;
    
    // Add user message immediately
    const userMessage: ChatMessage = {
      id: Date.now().toString(),
      reportId,
      role: 'user',
      content: text,
      timestamp: new Date()
    };
    setMessages(prev => [...prev, userMessage]);
    setInput('');
    setIsLoading(true);
    
    try {
      // Stream response
      const response = await fetch(`/api/nexus/reports/${reportId}/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message: text })
      });
      
      const aiMessage = await response.json();
      setMessages(prev => [...prev, aiMessage]);
    } catch (error) {
      console.error('Chat error:', error);
    } finally {
      setIsLoading(false);
    }
  };
  
  const clearChat = async () => {
    await fetch(`/api/nexus/reports/${reportId}/chat`, { method: 'DELETE' });
    setMessages([]);
  };
  
  return (
    <div className="flex flex-col h-full bg-[#12121a] rounded-lg border border-[#2d2d3a]">
      {/* Header */}
      <div className="flex items-center justify-between p-4 border-b border-[#2d2d3a]">
        <div className="flex items-center gap-2">
          <MessageCircle className="w-5 h-5 text-purple-500" />

        <button

          className="text-gray-400 hover:text-white transition-colors"
        >
          <RefreshCw className="w-4 h-4" />
        </button>
      </div>
      
      {/* Messages */}
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {messages.length === 0 ? (
          <div className="text-center text-gray-500 py-8">
            <MessageCircle className="w-12 h-12 mx-auto mb-3 opacity-50" />
            <p>დასვი კითხვა ანგარიშის შესახებ</p>
          </div>
        ) : (
          messages.map((msg) => (
            <ChatBubble key={msg.id} message={msg} />
          ))
        )}
        
        {isLoading && (
          <div className="flex items-center gap-2 text-gray-400">
            <Loader2 className="w-4 h-4 animate-spin" />
            <span>NEXUS ფიქრობს...</span>
          </div>
        )}
        
        <div ref={messagesEndRef} />

      {/* Quick Suggestions */}
      {messages.length === 0 && suggestions.length > 0 && (
        <div className="px-4 pb-2">
          <div className="flex items-center gap-2 mb-2 text-gray-400 text-sm">
            <Lightbulb className="w-4 h-4" />
            <span>სწრაფი კითხვები:</span>

              <button
                key={q.id}
                onClick={() => sendMessage(q.template)}
                className="px-3 py-1.5 text-sm bg-[#1a1a2e] text-gray-300 
                         rounded-full border border-[#2d2d3a] 
                         hover:border-purple-500 hover:text-white transition-all"
              >
                {q.icon} {q.label}
              </button>
            ))}
          </div>
        </div>
      )}
      
      {/* Input */}
      <div className="p-4 border-t border-[#2d2d3a]">

            value={input}

                     border border-[#2d2d3a] focus:border-purple-500

const ChatBubble: React.FC<{ message: ChatMessage }> = ({ message }) => {
  const isUser = message.role === 'user';

    <div className={`flex ${isUser ? 'justify-end' : 'justify-start'}`}>
      <div className={`max-w-[85%] rounded-lg p-3 ${

          ? 'bg-purple-600 text-white' 
          : 'bg-[#1a1a2e] text-gray-200 border border-[#2d2d3a]'
      }`}>
        {!isUser && (

  );
```
